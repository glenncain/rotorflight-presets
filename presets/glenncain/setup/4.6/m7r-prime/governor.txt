#$ TITLE: M7R Prime governor and throttle range

#$ FIRMWARE_VERSION: 4.6

#$ CATEGORY: SETUP
#$ STATUS: COMMUNITY
#$ KEYWORDS: m7r, omphobby, m7r prime, nexus xr, rotorflight, governor, electric, spoolup, throttle, rf gov
#$ AUTHOR: glenncain
#$ DESCRIPTION: The Rotorflight ELECTRIC governor driving a FUNCTION-type throttle output, with a
#$ DESCRIPTION: 300 ms spool-up, 100 ms tracking, and the bypass throttle curve.
#$ DESCRIPTION: Also sets the 1100-2000 throttle endpoints the governor works against.
#$ DESCRIPTION: Headspeed and the governor PID gains are NOT here - those live per-profile, in the
#$ DESCRIPTION: profile presets, so one governor setup can serve all three headspeeds.
#$ DESCRIPTION: Requires a working RPM source: apply the ESC link preset first.
#$ DESCRIPTION: Leaves PID gains, rates and filtering alone.
#$ PRIORITY: 0
#$ WARNING: Bench-test governor behaviour with blades off before flying it.

feature GOVERNOR
set gov_mode = ELECTRIC
set gov_throttle_type = FUNCTION
set gov_spoolup_time = 300
set gov_tracking_time = 100
set gov_auto_throttle = 50
set gov_bypass_throttle = 0,40,80,90,100,110,120,160,200
set rc_min_throttle = 1100
set rc_max_throttle = 2000
