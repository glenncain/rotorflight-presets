#$ TITLE: M7R Prime ESC link and RPM telemetry

#$ FIRMWARE_VERSION: 4.6
#$ BOARD_NAME: NEXUS_XR

#$ CATEGORY: SETUP
#$ STATUS: COMMUNITY
#$ KEYWORDS: m7r, omphobby, m7r prime, nexus xr, rotorflight, esc, dshot, omphobby esc, rpm, headspeed, uart
#$ AUTHOR: glenncain
#$ DESCRIPTION: DSHOT150 on the motor output, and ESC telemetry on UART2 using the OMPHOBBY
#$ DESCRIPTION: protocol in half-duplex with the pin swapped.
#$ DESCRIPTION: Enables ESC_SENSOR and FREQ_SENSOR, which is where headspeed comes from - the
#$ DESCRIPTION: governor and the RPM filters are both useless without it.
#$ DESCRIPTION: The UART2 assignment is specific to how this NEXUS_XR is wired.
#$ DESCRIPTION: Leaves the governor tune, filtering and PID profiles alone.
#$ PRIORITY: 0

feature ESC_SENSOR
feature FREQ_SENSOR
serial 1 1024 115200 57600 0 115200
set use_unsynced_pwm = OFF
set motor_pwm_protocol = DSHOT150
set esc_sensor_protocol = OMPHOBBY
set esc_sensor_halfduplex = ON
set esc_sensor_pinswap = ON
